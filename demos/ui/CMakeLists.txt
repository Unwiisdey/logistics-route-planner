cmake_minimum_required(VERSION 3.16)
project(LogisticsUIDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DEFAULT_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LOGISTICS_PROJECT "${DEFAULT_PROJECT}" CACHE PATH "Existing course project source")
set(BUILD_TESTING OFF CACHE BOOL "Disable unrelated tests in the standalone demo build" FORCE)
add_subdirectory("${LOGISTICS_PROJECT}" "${CMAKE_CURRENT_BINARY_DIR}/core" EXCLUDE_FROM_ALL)
add_executable(demo_bridge bridge.cpp)
target_include_directories(demo_bridge PRIVATE "${LOGISTICS_PROJECT}/include" "${LOGISTICS_PROJECT}/third_party")
target_link_libraries(demo_bridge PRIVATE logistics_core)
if(MSVC)
  target_compile_options(demo_bridge PRIVATE /W4 /permissive- /utf-8)
endif()
